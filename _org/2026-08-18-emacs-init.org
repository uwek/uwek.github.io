  ;; Indirekten Buffer für Subtree mit Heading-Präfix erzeugen (narrowed).
  (defun my/org-subtree-to-indirect (prefix)
    "Erzeuge oder hole indirekten Buffer für Subtree mit Heading PREFIX.
  Gibt den Buffer zurück oder nil wenn kein passendes Heading gefunden."
    (let* ((base-buf (current-buffer))
           (buf-name (format "%s::%s" (buffer-name base-buf) prefix)))
      (save-excursion
        (save-restriction
          (widen)
          (goto-char (point-min))
          (when (re-search-forward
                 (concat "^\\*+ " (regexp-quote prefix)) nil t)
            (beginning-of-line)
            (let* ((heading-pos (point))
                   (ibuf (or (get-buffer buf-name)
                             (make-indirect-buffer base-buf buf-name t))))
              (with-current-buffer ibuf
                (widen)
                (goto-char heading-pos)
                (org-narrow-to-subtree))
              ibuf))))))

  ;; Subtree in Side-Window anzeigen — explizites display-buffer, kein Namens-Matching.
  (defun my/scrivener-toggle (prefix side slot &optional width)
    "Toggle Side-Window für Subtree mit Heading PREFIX."
    (if-let* ((ibuf (my/org-subtree-to-indirect prefix))
              (win  (get-buffer-window ibuf)))
        (delete-window win)
      (when-let ((ibuf (my/org-subtree-to-indirect prefix))
                 (win  (display-buffer ibuf
                                       `(display-buffer-in-side-window
                                         (side . ,side)
                                         (slot . ,slot)
                                         (window-width . ,(or width 0.25))
                                         (preserve-size . (t . nil))
                                         (window-parameters . ((no-delete-other-windows . t)))))))
        (set-window-point win (with-current-buffer ibuf (point-min)))
